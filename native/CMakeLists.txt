cmake_minimum_required(VERSION 3.20)
project(activity_infer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT ONNXRUNTIME_ROOT)
  message(FATAL_ERROR "Set ONNXRUNTIME_ROOT to the extracted ONNX Runtime package")
endif()
add_executable(activity_infer infer.cpp)
target_include_directories(activity_infer PRIVATE "${ONNXRUNTIME_ROOT}/include")
find_library(ORT_LIBRARY onnxruntime PATHS "${ONNXRUNTIME_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
target_link_libraries(activity_infer PRIVATE "${ORT_LIBRARY}")
if(WIN32)
  add_custom_command(TARGET activity_infer POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${ONNXRUNTIME_ROOT}/lib/onnxruntime.dll" "$<TARGET_FILE_DIR:activity_infer>")
endif()
